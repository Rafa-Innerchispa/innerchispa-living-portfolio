        async function triggerAutoSubmit() {
            const url = document.getElementById('app-modal-url').value;
            const btn = document.getElementById('btn-auto-submit');
            const statusBox = document.getElementById('app-submit-status-box');
            
            btn.disabled = true;
            btn.classList.add('opacity-50', 'cursor-not-allowed');
            statusBox.classList.remove('hidden');
            statusBox.innerHTML = `Iniciando agente Playwright para el envío automatizado a la plataforma del evento...`;
            
            try {
                const res = await fetch('/api/hackathons/auto-submit', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ url: url })
                });
                const data = await res.json();
                if (res.ok) {
                    startSubmitPolling(url);
                } else {
                    statusBox.innerHTML = `Error: ${data.message}`;
                    btn.disabled = false;
                    btn.classList.remove('opacity-50', 'cursor-not-allowed');
                }
            } catch (err) {
                statusBox.innerHTML = `Error de red al conectar con el servidor.`;
                btn.disabled = false;
                btn.classList.remove('opacity-50', 'cursor-not-allowed');
            }
        }

        function startSubmitPolling(url) {
            if (submitPollInterval) clearInterval(submitPollInterval);
            
            const btn = document.getElementById('btn-auto-submit');
            const statusBox = document.getElementById('app-submit-status-box');
            
            submitPollInterval = setInterval(async () => {
                try {
                    const res = await fetch(`/api/hackathons/auto-submit/status?url=${encodeURIComponent(url)}`);
                    const data = await res.json();
                    
                    if (data.status === 'success') {
                        clearInterval(submitPollInterval);
                        submitPollInterval = null;
                        btn.disabled = false;
                        btn.classList.remove('opacity-50', 'cursor-not-allowed');
                        
                        let screenshotEl = '';
                        if (data.screenshot) {
                            screenshotEl = `Ver Captura de Pantalla del Envío`;
                        }
                        statusBox.innerHTML = `
                            ¡Envío completado con éxito!
                            ${data.message}
                            ${screenshotEl}
                        `;
                    } else if (data.status === 'error') {
                        clearInterval(submitPollInterval);
                        submitPollInterval = null;
                        btn.disabled = false;
                        btn.classList.remove('opacity-50', 'cursor-not-allowed');
                        statusBox.innerHTML = `
                            Error al enviar:
                            ${data.message}
                            Ver Captura de Pantalla del Error
                        `;
                    } else {
                        statusBox.innerHTML = `Estado: ${data.message || 'Procesando...'}`;
                    }
                } catch (err) {
                    console.error("Error polling submit status:", err);
                }
            }, 3000);
        }

        function toggleVoiceInput() {
            const input = document.getElementById('ai-prompt-input');
            const micBtn = document.getElementById('btn-voice-input');
            const headerMicBtn = document.getElementById('btn-voice-input-header');
            
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition || window.mozSpeechRecognition || window.msSpeechRecognition;
            
            if (!SpeechRecognition) {
                alert("Tu navegador o sistema operativo no soporta el reconocimiento de voz nativo en este origen. Si estás usando Chrome sobre HTTP (no localhost/HTTPS), debes configurarlo primero.");
                return;
            }
            
            if (isListening) {
                if (voiceRecognition) voiceRecognition.stop();
                return;
            }
            
            voiceRecognition = new SpeechRecognition();
            voiceRecognition.lang = 'es-ES';
            voiceRecognition.continuous = true;
            voiceRecognition.interimResults = false;
            
            voiceRecognition.onstart = () => {
                isListening = true;
                [micBtn, headerMicBtn].forEach(btn => {
                    if (btn) {
                        btn.classList.remove('bg-white/5');
                        btn.classList.add('bg-purple-600', 'text-white', 'animate-pulse');
                        btn.title = "Detener micrófono";
                    }
                });
            };
            
            voiceRecognition.onerror = (e) => {
                console.error("Error en reconocimiento de voz:", e);
                stopVoiceRecognition();
                
                if (e.error === 'not-allowed') {
                    alert("Acceso al micrófono denegado.\n\n" +
                          "Nota importante de seguridad: Los navegadores modernos bloquean el micrófono en conexiones HTTP no seguras.\n" +
                          "Para solucionarlo en Chrome, visita:\n" +
                          "  chrome://flags/#unsafely-treat-insecure-origin-as-secure\n\n" +
                          "Habilita la opción e ingresa tu URL actual: http://" + window.location.host);
                } else if (e.error === 'no-speech') {
                    // Quiet failure, just stop
                } else {
                    alert("Error de micrófono (" + e.error + "). Asegúrate de tener un micrófono activo.");
                }
            };
            
            voiceRecognition.onend = () => {
                stopVoiceRecognition();
            };
            
            voiceRecognition.onresult = (event) => {
                let finalTranscripts = '';
                for (let i = event.resultIndex; i < event.results.length; ++i) {
                    if (event.results[i].isFinal) {
                        finalTranscripts += event.results[i][0].transcript + ' ';
                    }
                }
                if (finalTranscripts.trim()) {
                    input.value = (input.value ? input.value + " " : "") + finalTranscripts.trim();
                    input.focus();
                    scrollToBottom();
                }
            };
            
            try {
                voiceRecognition.start();
            } catch (err) {
                console.error("No se pudo iniciar el servicio de voz:", err);
                alert("Error de seguridad o de origen al iniciar el micrófono.");
            }
        }

        async function renameConversationPrompt(convId, currentTitle) {
            const newTitle = prompt("Renombrar conversación:", currentTitle);

            const trimmed = newTitle.trim();
            if (!trimmed) return;
            try {
                const res = await fetch(`/api/ai/conversations/${convId}/rename`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },

            const gridSize = 100;
            for (let x = 0; x < width; x += gridSize) {
                ctx.beginPath();
                ctx.moveTo(x, 0);
                ctx.lineTo(x, height);
                ctx.stroke();
            }
            for (let y = 0; y < height; y += gridSize) {
                ctx.beginPath();
                ctx.moveTo(0, y);
                ctx.lineTo(width, y);
                ctx.stroke();
            }

            // Update particle positions
            for (let i = 0; i < bgParticles.length; i++) {
                const p1 = bgParticles[i];
                p1.x += p1.vx;
                p1.y += p1.vy;
                
                // Boundaries check
